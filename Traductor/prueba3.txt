int[] numeros = new int[] { 3, 6, 9, 12 };
int suma = 0;
int encontrados = 0;

for (int i = 0; i < numeros.Length; i++)
{
    if (numeros[i] < 6)
    {
        continue;
    }

    if (numeros[i] > 10)
    {
        break;
    }

    suma += numeros[i];
    encontrados++;
}

foreach (int numero in numeros)
{
    if (numero == 9)
    {
        Console.WriteLine("Encontrado");
    }
}

Console.WriteLine(suma);
Console.WriteLine(encontrados);
